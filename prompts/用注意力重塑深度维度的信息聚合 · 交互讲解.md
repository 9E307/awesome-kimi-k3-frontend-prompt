## 4. 全局交互与动效设计（必须逐条实现）					
					
每个动画章节都具有独立播放器状态：					
- `steps`：该章节拍点数组，每拍含 `label` 与异步 `go()`。					
- `idx`：下一拍索引；`playing`：自动播放状态；`busy`：当前拍动画是否仍在执行；`gen`：重置时递增，用来让旧异步流程失效。					
- “播放”变为“暂停”；暂停只阻止下一拍开始，不粗暴打断当前正在插值的这一拍。					
- “单步”只执行一个完整拍点；当前拍未完成时忽略额外单步点击，防止竞态。					
- “重置”必须清空并重建该章节 SVG，恢复说明、按钮、公式、卡片和所有隐藏元素。					
- 动画播完后说明栏显示“完成 / 演示结束，点击‘重置’可重播”。					
- 鼠标按下某章节后，该章节成为键盘控制目标。默认活动章节是 Section 3。					
- 键盘：空格=播放/暂停，右箭头=单步，R=重置；焦点在 select/input/textarea 内时不拦截。

const NS = 'http://www.w3.org/2000/svg';					
const $ = id => document.getElementById(id);					
function S(tag, attrs, parent){					
  const e = document.createElementNS(NS, tag);					
  if(attrs) for(const k in attrs) e.setAttribute(k, attrs[k]);					
  if(parent) parent.appendChild(e);					
  return e;

function txt(parent, x, y, str, attrs){					
  const t = S('text', Object.assign({x, y, 'font-size':13, fill:'#4e4e4e'}, attrs||{}), parent);					
  t.textContent = str;					
  return t;					
}					
let SPEED = 1;					
const sleep = ms => new Promise(r => setTimeout(r, Math.max(1, ms/SPEED)));					
/* 用 CSS transition 做插值；对 SVG 元素仅使用 opacity / transform / stroke-dashoffset 等可过渡属性 */					
function css(el, style, ms){					
  return new Promise(res => {					
    el.style.transition = ms ? `all ${ms/SPEED}ms cubic-bezier(.4,0,.2,1)` : 'none';					
    requestAnimationFrame(() => requestAnimationFrame(() => Object.assign(el.style, style)));					
    setTimeout(res, (ms ? ms/SPEED : 0) + 40);					
  });					
}					
function pulse(el, ms=380){					
  el.style.transition = 'none'; el.style.opacity = '0.55';					
  requestAnimationFrame(()=>requestAnimationFrame(()=>{					
    el.style.transition = `opacity ${ms/SPEED}ms ease`; el.style.opacity = '0';					
  }));					
  return sleep(ms);					
}					
/* 沿路径流动的小球 */					
function flowPacket(path, color, r, dur){					
  return new Promise(res => {					
    let len; try { len = path.getTotalLength(); } catch(e){ res(); return; }					
    if(!len){ res(); return; }					
    const c = S('circle', {r:r||5, fill:color, stroke:'#fff', 'stroke-width':1.5}, path.ownerSVGElement);					
    const t0 = performance.now();					
    (function frame(t){					
      if(!c.isConnected){ res(); return; }					
      const p = Math.min(1, (t-t0)/(dur/SPEED));					
      const pt = path.getPointAtLength(p*len);					
      c.setAttribute('cx', pt.x); c.setAttribute('cy', pt.y);					
      if(p < 1) requestAnimationFrame(frame);					
      else { c.remove(); res(); }					
    })(t0);					
  });					
}					
/* 路径描边动画 */					
function drawPath(path, ms){					
  let len; try { len = path.getTotalLength(); } catch(e){ return Promise.resolve(); }					
  path.style.transition = 'none';					
  path.style.strokeDasharray = len; path.style.strokeDashoffset = len;					
  return new Promise(res => {					
    requestAnimationFrame(()=>requestAnimationFrame(()=>{					
      path.style.transition = `stroke-dashoffset ${ms/SPEED}ms ease`;					
      path.style.strokeDashoffset = '0';					
    }));					
    setTimeout(res, ms/SPEED + 40);					
  });					
}

}					
					
/* ===================== 播放器 ===================== */					
let activeSec = null;					
function setActive(sec){ activeSec = sec; }					
function initSection(num, buildFn){					
  const sec = {					
    num, steps:[], idx:0, playing:false, busy:false, gen:0, gateRes:null,					
    ui:{ svg:$('svg'+num), cap:$('cap'+num), play:$('play'+num), step:$('step'+num), reset:$('reset'+num) }					
  };					
  sec.update = () => {					
    sec.ui.play.textContent = sec.playing ? '⏸ 暂停' : '▶ 播放';					
    sec.ui.step.disabled = sec.playing || sec.idx >= sec.steps.length;					
    if(sec.idx >= sec.steps.length && !sec.busy)					
      sec.ui.cap.innerHTML = '<span class="n">完成</span>演示结束，点击“重置”可重播';					
  };					
  sec.rebuild = () => {					
    sec.ui.svg.innerHTML = '';					
    sec.steps = buildFn(sec.ui.svg, sec) || [];					
    sec.idx = 0;					
    sec.ui.cap.innerHTML = '<span class="n">提示</span>点击“播放”自动演示，或“单步”逐拍讲解（空格=播放/暂停，→=单步，R=重置）';					
    renderAllTex(sec.ui.svg);					
    sec.update();					
  };					
  sec.gate = () => sec.playing ? Promise.resolve() : new Promise(r => sec.gateRes = r);					
  sec.loop = async (g) => {					
    sec.busy = true;					
    while(sec.idx < sec.steps.length){					
      if(g !== sec.gen) { sec.busy = false; return; }					
      const st = sec.steps[sec.idx];					
      sec.ui.cap.innerHTML = `<span class="n">${sec.idx+1}/${sec.steps.length}</span>${st.label}`;					
      try { await st.go(g); } catch(e){ console.warn(e); }					
      if(g !== sec.gen) { sec.busy = false; return; }					
      sec.idx++;					
      sec.update();					
      if(sec.idx < sec.steps.length) await sec.gate();					
      if(g !== sec.gen) { sec.busy = false; return; }					
    }					
    sec.playing = false; sec.busy = false; sec.update();					
  };					
  sec.ui.play.onclick = () => {					
    setActive(sec);					
    if(sec.playing){ sec.playing = false; sec.update(); return; }					
    if(sec.idx >= sec.steps.length){ sec.gen++; sec.rebuild(); }					
    sec.playing = true; sec.update();					
    if(sec.gateRes){ const r = sec.gateRes; sec.gateRes = null; r(); }					
    if(!sec.busy) sec.loop(sec.gen);					
  };					
  sec.ui.step.onclick = () => {					
    setActive(sec);					
    if(sec.playing){ sec.playing = false; sec.update(); return; }					
    if(sec.idx >= sec.steps.length) return;					
    if(sec.gateRes){ const r = sec.gateRes; sec.gateRes = null; r(); }					
    else if(!sec.busy) sec.loop(sec.gen);					
    // busy 且无挂起 gate = 动画进行中，忽略本次点击					
  };					
  sec.ui.reset.onclick = () => {					
    setActive(sec);					
    sec.gen++; sec.playing = false; sec.busy = false;					
    if(sec.gateRes){ const r = sec.gateRes; sec.gateRes = null; r(); }					
    sec.rebuild();					
  };					
  sec.ui.svg.closest('section').addEventListener('mousedown', () => setActive(sec));					
  sec.rebuild();					
  return sec;					
}
